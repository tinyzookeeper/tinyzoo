export interface AnalyticsOptions {
    keepRawDays?: number;
    access?: (ctx: {
        user: CmsAccessUser;
    }) => boolean;
    respectDoNotTrack?: boolean;
    countryHeaders?: string[];
    geoipFile?: string;
    timeZone?: string;
    excludePaths?: string[];
}

export interface AnalyticsRollupDay {
    day: string;
    visitors: number;
    visits: number;
    pageviews: number;
    bounces: number;
    engagedMs: number;
    pages: AnalyticsRollupItem[];
    sources: AnalyticsRollupItem[];
    countries: AnalyticsRollupItem[];
    devices: AnalyticsRollupItem[];
}

export interface AnalyticsRollupItem {
    value: string;
    title?: string;
    visitors: number;
    visits: number;
    pageviews: number;
}

export interface AnalyticsRollups {
    format: 'tinyzoo-analytics-rollups';
    version: 1;
    at: string;
    timeZone: string;
    from: string;
    to: string;
    since: string | null;
    through: string | null;
    days: AnalyticsRollupDay[];
}

export interface DailyRollup {
    day: string;
    dimension: string;
    value: string;
    visitors: number;
    visits: number;
    pageviews: number;
    bounces: number;
    engagedMs: number;
}

export function analyticsPlugin(options?: AnalyticsOptions): CmsPlugin;

export function dailyRollups(db: any, from: string, to: string, dimensions?: readonly string[]): Promise<DailyRollup[]>;
