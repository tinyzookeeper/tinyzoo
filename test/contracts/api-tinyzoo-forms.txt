export interface FormOptions<F extends Record<string, any>> {
    label: LocalizedText;
    labelSingular?: LocalizedText;
    description?: LocalizedText;
    fields: F;
    titleField?: keyof F & string;
    submitLabel?: LocalizedText;
    successMessage?: LocalizedText;
    notify?: {
        to?: string[];
    };
    retentionDays?: number;
    access?: EntityDef<F>['access'];
}

export type FormStatusField = FieldDescriptor<{
    out: 'new' | 'read' | 'done';
    in: 'new' | 'read' | 'done';
    col: 'scalar';
    optionalIn: true;
}>;

export function defineForm<F extends Record<string, AnyFieldDescriptor>>(name: string, opts: FormOptions<F>): Entity<F & {
    status: FormStatusField;
}>;

export function defineFormSettings(opts?: {
    name?: string;
    access?: EntityDef<any>["access"];
}): Entity<Record<string, AnyFieldDescriptor>>;

export function formsPlugin(): CmsPlugin;
